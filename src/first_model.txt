from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense, Dropout


def build_model():
    model = Sequential([
        Conv2D(32, (3, 3), activation='relu', input_shape=(224, 224, 3)),
        MaxPooling2D(2, 2),

        Conv2D(64, (3, 3), activation='relu'),
        MaxPooling2D(2, 2),

        Flatten(),
        Dense(128, activation='relu'),
        Dropout(0.5),

        # CHANGED: 6 classes (LEVEL 2 to LEVEL 7)
        # Activation is softmax for multi-class classification
        Dense(6, activation='softmax')
    ])

    # CHANGED: Use 'sparse_categorical_crossentropy' because labels are integers (0, 1, 2...)
    # If you used One-Hot (vectors), you would use 'categorical_crossentropy'
    model.compile(optimizer='adam',
                  loss='sparse_categorical_crossentropy',
                  metrics=['accuracy'])

    return model
