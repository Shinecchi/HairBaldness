from tensorflow.keras.applications import MobileNetV2
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Dense, GlobalAveragePooling2D, Dropout
from tensorflow.keras.optimizers import Adam


def build_model():
    # 1. Load the pre-trained MobileNetV2 model
    # include_top=False removes the final classification layer so we can add our own
    base_model = MobileNetV2(weights='imagenet', include_top=False, input_shape=(224, 224, 3))

    # 2. Freeze the base model
    # This prevents the pre-learned patterns from being destroyed during initial training
    base_model.trainable = False

    # 3. Add new layers for our specific 6 classes
    x = base_model.output
    x = GlobalAveragePooling2D()(x)
    x = Dense(128, activation='relu')(x)
    x = Dropout(0.5)(x)  # Helps prevent overfitting
    predictions = Dense(6, activation='softmax')(x)  # 6 Output classes

    # 4. Compile
    model = Model(inputs=base_model.input, outputs=predictions)

    # We use a lower learning rate (0.0001) for fine-tuning
    model.compile(optimizer=Adam(learning_rate=0.0001),
                  loss='sparse_categorical_crossentropy',
                  metrics=['accuracy'])

    return model
